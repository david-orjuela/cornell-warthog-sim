start=0
end=1985195360000

find . -maxdepth 2 -type f -name '*.log' -print0 |
while IFS= read -r -d '' file; do
    if [[ "$file" =~ _([0-9]{13})\.log$ ]]; then
        ms="${BASH_REMATCH[1]}"

        if (( ms >= start && ms <= end )); then
            grep -HniE \
              'outdoor_tree_6' \
              "$file"
        fi
    fi
done
